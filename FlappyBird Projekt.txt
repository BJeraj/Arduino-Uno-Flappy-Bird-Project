void loop() {
  if(digitalRead(A0) == HIGH){
     a++;
  } 
  
  if(a >=1){
      if(a == 1){
      zacetek();
      a++;
    }
    
    if(tmr < 8){  //"gravitacija"
      tmr++;
      delay75();
    }
    if(a == 2){
      tmp = 8;
      
    }
    if(digitalRead(A0) == HIGH){  //jump
      tmr = tmr - 1;
      delay40();
      player();
      pipes();
      tmr = tmr - 1;
      delay40();
      player();
      pipes();
      tmr = tmr - 1;
      delay40();
      player();
      pipes(); 

      }else {
      player();
    }
    pipes();
    reset();
    if(a == 2){
      tmp = 8;
      a++;
    } 
    //-----------------------------------------------
    if(tmp == 0){
      tmp = 8;
      rdm = 1 + (rand() % 4);
      }else{
        if(tmp >= 1 ){
        tmp = tmp - 1;
        delay100();
        pipes();    
        player();
    
      }
    }
    smrt();
  }
}
